const _out = 'docs/design/baseline/G3_2026-09-29/preview';

Future<void> _loadFonts() async {
  final pretendard = FontLoader('Pretendard');
  for (final w in ['Regular', 'Medium', 'SemiBold', 'Bold', 'ExtraBold']) {
    pretendard.addFont(Future.value(ByteData.sublistView(File('assets/fonts/Pretendard-$w.ttf').readAsBytesSync())));
  }
  await pretendard.load();
  final icons = FontLoader('MaterialIcons')
    ..addFont(Future.value(ByteData.sublistView(File(
        'C:/flutter_windows_3.41.5-stable/flutter/bin/cache/artifacts/material_fonts/materialicons-regular.otf')
        .readAsBytesSync())));
  await icons.load();
}

Widget _caption(String s) => Padding(
  padding: const EdgeInsets.only(top: 18, bottom: 8),
  child: Text(s, style: const TextStyle(fontFamily: 'Pretendard', fontSize: 12, fontWeight: FontWeight.w700, color: MLColors.textSoft, letterSpacing: 0.4)),
);

Future<void> _shoot(WidgetTester tester, String name, List<Widget> children) async {
  await tester.binding.setSurfaceSize(const Size(411, 2600));
  const key = Key('shot');
  await tester.pumpWidget(MaterialApp(
    debugShowCheckedModeBanner: false,
    theme: AppTheme.lightTheme,
    home: Scaffold(body: SingleChildScrollView(child: RepaintBoundary(
      key: key,
      child: Container(
        color: MLColors.bg,
        padding: const EdgeInsets.fromLTRB(20, 4, 20, 24),
        child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, mainAxisSize: MainAxisSize.min, children: children),
      ),
    ))),
  ));
  await tester.pumpAndSettle();
  final boundary = tester.renderObject<RenderRepaintBoundary>(find.byKey(key));
  final image = await tester.runAsync(() => boundary.toImage(pixelRatio: 2));
  final bytes = await tester.runAsync(() => image!.toByteData(format: ui.ImageByteFormat.png));
  Directory(_out).createSync(recursive: true);
  File('$_out/$name.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  debugDisableShadows = true;
}
